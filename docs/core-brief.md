- struct CoastTripItem: Codable, Identifiable { id:String UUID; activityID:String; day:Int=0; titleSnapshot:String; }
- struct CoastEntry: Codable, Identifiable { id:String UUID; title:String=""; body:String=""; date:String yyyy-MM-dd; tripID:String?; photos:[String]=[]; isDraft:Bool=true }
- struct CoastProgress: Codable { step:Int=0; completed:Bool=false }
- struct CoastLedger: Codable { trips:[CoastTrip]=[]; entries:[CoastEntry]=[]; bookmarks:Set<String>=[]; progress:[String:CoastProgress]=[:] }
- enum CoastValidation: static func trip(_ value:CoastTrip) -> String?; static func entry(_ value:CoastEntry) -> String?; static func email(_ value:String)->Bool; static func password(_ value:String)->Bool. Error strings are stable English keys that UI translates. Limits exactly field contract.
